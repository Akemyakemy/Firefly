## 搜索

搜索是通过穷尽所有的可能来找到最优解，或者统计合法解的个数的算法。

### BFS算法

BFS算法（广度优先搜索）是一种图搜索算法，它从根节点开始，按照**层序遍历**的顺序搜索图中的节点。

```c++
vector<int> e[N];//邻接表，假设目标是从s走到t
int target=t;//目标

bool bfs(int s){//广搜查看s是否能到达t
    queue<int> q;
    q.push(s);
    while(q.size()){
        int u=q.front();
        if(u==target) return true;
        q.pop();
        for(auto v:e[u]) q.push(v);
    }
    return false;
}
```

#### 双向BFS算法

从始态和终态两个状态同时进行BFS，当**两个搜索方向相遇**时立即返回结果。

```c++
vector<int> e[N];//邻接表，假设目标是从s走到t

bool bfs(int s,int t){
    if(s==t) return true;
    bool vis1[N]={false},vis2[N]={false};
    queue<int> a,b;
    a.push(s),b.push(t);
    vis1[s]=1,vis2[t]=1;
    while(1){
        if(a.empty() || b.empty()) return false;
        int u1=a.front(),u2=b.front();
        a.pop();b.pop();
        for(auto v:e[u1]){
            if(vis1[v]) continue;
            if(vis2[v]) return true;
            else{
                a.push(v);vis1[v]=1;
            }
        }
        for(auto v:e[u2]){
            if(vis2[v]) continue;
            if(vis1[v]) return true;
            else{
                b.push(v);vis2[v]=1;
            }
        }
    }
}
```

#### 双端队列BFS算法

又叫做**0-1BFS**。

在**只存在两种边权**的情况下，求s到t的最短路径时使用，小边权放队首，大边权放队尾。

```c++
#define pii pair<int,int>
#define v first
#define w second
const int inf=LLONG_MAX;
vector<pii> e[N];//{v,w},v是终点编号，w是边权
int target=t;
int dis[N];//存储任何点到s的最短距离

bool bfs(int s){
    for(int i=1;i<=n;i++) dis[i]=inf;
    deque<int> q;
    q.push_front(s);dis[s]=0;
    while(q.size()){
        int u=q.front();q.pop_front();//保证优先处理短边权的路径
       	if(u==target) return true;
        for(auto ed:e[u]){
            int v=ed.v,w=ed.w;
            if(dis[v]>dis[u]+w){
                dis[v]=dis[u]+w;
                if(q.empty() || w<=q.front().y) q.push_front(v);//小边权优先
                else q.push_back(v);
            }
        }
    }
}

signed main(){
    if(bfs(s)) cout<<dis[e]<<endl;
    else puts("-1");
}
```

### DFS算法

DFS算法（深度优先搜索）是一种图搜索算法，它从根节点开始，按照**先序遍历**的顺序搜索图中的节点。

```c++
vector<int> e[N];//邻接表，假设目标是从s走到t
int target=t;//目标

bool dfs(int u){
    if(u==target) return true;
    bool ans=false;
    for(auto v:e[u]){
        ans|=dfs(v);
        if(ans) break;//只要有一条路走通就直接返回了
    }
    return ans;
}
```

#### 迭代加深搜索

DFS算法如果开始选错了分支，就会浪费大量时间。迭代加深搜索是一种优化的DFS算法，它会先搜索深度较浅的分支，直到找到目标节点或搜索完所有节点。

```c++
vector<int> e[N];//邻接表，假设目标是从s走到t
int target=t;//目标

bool dfs(int u,int d,int limit){//编号与深度和深度极限
    if(d==limit) return u==target;
    bool ans=false;
    for(auto v:e[u]){
        ans|=dfs(v);
        if(ans) break;
    }
    return ans;
}

signed main(){
    for(int i=1;i<=(int)log(n)+1;i+) if(dfs(s,1,i)){
        puts("yes");
        return 0;
    }
    puts("no");
}
```

#### 双向DFS算法

从始态和终态两个状态**分别而非同时**进行DFS，分别只搜所有节点的一半，然后碰头返回结果。

```c++
vector<int> e[N];//邻接表，假设目标是从s走到t
bool vis[N];

void dfs1(int u,int d){//当前节点和层数
    if(d==(log(n)+1)/2) vis[u]=1,return;
    for(auto v:e[u]) dfs(v,d+1);
}

bool dfs2(int u){
    if(vis[u]) return true;
    bool ans=false;
    for(auto v:e[u]){
        ans|=dfs2(v);
        if(ans) break;
    }
    return ans;
}

signed main(){
    dfs1(s,1);
    if(dfs2(t)) puts("yes");
    else puts("no");
}
```

### A*算法

带有**估价函数的优先队列BFS算法**就是A*算法。是[Dijkstra算法](https://blog.542000.xyz/posts/%E5%9B%BE%E8%AE%BA/%E6%9C%80%E7%9F%AD%E8%B7%AF%E5%BE%84%E7%AE%97%E6%B3%95/)的进一步优化。

与传统BFS的区别，利用优先队列维护**当前代价+未来代价最小的状态**。

估价函数：$f(x) = g(x) + h(x)$

其中，$g(x)$ 是当前代价，$h(x)$ 是未来代价。由于 $h(x)$ 未知，因此需要一种方法对其进行简单估算。当$h(x)=0$时退化成Dijkstra算法。

一种可以用于**求解第K短路**的算法。

示例：从s到t第k短的路径长度。

1. 预处理$f(x)$ 为以$t$为起点的Dijkstra算法最短路径的路径长度。这里的估价函数值等于实际值。
2. 利用优先队列跑BFS，队中的节点按照`dis[v]+f[v]`自动升序排列。
3. 允许每个点最多入队$k$次，那么当$T$第$k$次出队时，即为第$k$短路。

$Dijkstra\ O(nlogn)$+$优先队列BFS\ O((k+m)log(k+m))$

```c++
#include<bits/stdc+.h>
using namespace std;
#define int long long
#define pii pair<int,int>
#define x first
#define y second
const int N=1e5+9;//n,m均<=1e5
const int inf=LLONG_MAX;

int n,m,s,t,k;

struct edge{
	int v,w;	
};
vector<edge> e[N];
int f[N],vis[N];
void dijkstra(int t){
	for(int i=1;i<=n;i++) f[i]=inf;
	priority_queue<pii,vector<pii>,greater<pii>> q;//小根堆
	q.push_back({0,t});
	while(q.size()){
		int u=q.top().y;q.pop();
		if(vis[u]) continue;
		vis[u]=1;
		for(auto ed:e[u]){
			int v=ed.v,w=ed.w;
			if(f[v]>f[u]+w){
				f[v]=f[u]+w;
				q.push({f[v],v});
			}
		}
	}
}

struct node{
	int d,u,l;//d为l+f[v],l是s到u的实际距离
	const bool operator < (const node& a){
		return d>a.d;
	}
};
int cnt[N],ans;
bool aStar(){
    if(f[s]==inf) return false;
	if(s==t) k++;//重合点，长度为0是第一条
	priority_queue<node> q;//注意重载函数的符号，此处是小根堆
	node a={f[s],s,0};q.push(a);
	while(q.size()){
		int u=q.top().u,l=q.top().l;q.pop();
		cnt[u]++;
		if(cnt[t]==k){
			ans=l;
			return true;
		}
		for(int ed:e[u]){
			int v=ed.v,w=ed.w;
			if(cnt[v]<k){
				node a={l+w+f[v],v,l+w};
				q.push(a);
			}
		}
	}
}

signed main(){
	cin.tie(nullptr)->sync_with_stdio(false);
	cin>>n>>m;
	for(int i=1,,u,v,w;i<=m;i++){
		cin>>u>>v>>w;
		e[u].push_back({v,w});
		e[v].push_back({u,w});
	}
	cin>>s>>t>>k;
	dijkstra(t);
	if(aStar()) cout<<ans<<endl;
	else puts("-1");
	return 0;	
}
```

### IDA*算法

A*算法=估价函数+优先队列BFS

IDA*算法=A*算法+迭代加深搜索（IDFS）

如果$当前深度d+未来估价步数f(x)>深度限制limit$，那么立即回溯。

程序实现的代码量一般少于A*算法。

**估价函数**的值一定要满足：未来的估价$f(x)$<=实际未来代价g(x)。

如果$f(x)=0$,那么退化成普通的IDFS算法。

如果$f(x)>g(x)$,对于BFS可能会把最优分支的状态排在队伍后面，从而得出错误答案；对于DFS可能会提前终止搜索，从而找不到答案。

IDA*算法实现第k短路。

1. 每次迭代依据，bound上界的大小，初始值为s到t的最短路。
2. 每次小幅提升上界的大小保证搜索不爆内存。
3. 当上界的大小正好提升到第k短路径的大小时，一定能在第k次捕获这个值。

```c++
#include<bits/stdc++.h>
using namespace std;
#define int long long
#define pii pair<int,int>
const int N = 1e5+9;
const int inf = LLONG_MAX;

int n, m, s, t, k;
struct edge { int v, w; };
vector<edge> e[N];
int f[N], vis[N];           // f[u] = 从 u 到 t 的最短距离

// 反向 Dijkstra 计算启发函数
void dijkstra(int t) {
    for(int i = 1; i <= n; i++) f[i] = inf;
    priority_queue<pii, vector<pii>, greater<pii>> q;
    q.push({0, t}); f[t] = 0;
    while(q.size()) {
        int u = q.top().second; q.pop();
        if(vis[u]) continue;
        vis[u] = 1;
        for(auto ed : e[u]) {
            int v = ed.v, w = ed.w;
            if(f[v] > f[u] + w) {
                f[v] = f[u] + w;
                q.push({f[v], v});
            }
        }
    }
}

int cnt[N];        // 当前 DFS 路径中每个节点被访问的次数（回溯时恢复）
int ans;           // 最终答案
int found;         // 是否已经找到第 k 条路径
int target_cnt;    // 当前 bound 下已经到达 t 的次数
int bound;         // 当前迭代的代价上界
int next_bound;    // 下一次迭代的候选上界

// 深度优先搜索
// u: 当前节点, g: 从 s 到 u 的实际距离
void dfs(int u, int g) {
    if(found) return;                     // 已经找到答案，提前结束
    int h = f[u];                         // 估计剩余代价
    if(g + h > bound) {                   // 超过上界，剪枝
        next_bound = min(next_bound, g + h); // 记录下一次可能的上界
        return;
    }
    if(u == t) {                          // 到达终点
        target_cnt++;                     // 找到一条新路径
        if(target_cnt == k) {             // 是第 k 条
            ans = g;
            found = 1;
        }
        return;
    }
    if(cnt[u] >= k) return;               // 防止无限循环：一个节点重复太多次
    cnt[u]++;                             // 进入节点 u
    for(auto ed : e[u]) {
        int v = ed.v, w = ed.w;
        dfs(v, g + w);                    // 继续深入
        if(found) return;
    }
    cnt[u]--;                             // 回溯
}

bool ida_star() {
    if(s == t) k++;                       // 起点等于终点时，空路径算第一条
    dijkstra(t);                          // 计算启发函数
    if(f[s] == inf) return false;         // s 无法到达 t

    bound = f[s];                         // 初始上界 = 最短路径长度
    while(true) {
        target_cnt = 0;
        found = 0;
        next_bound = inf;
        memset(cnt, 0, sizeof(cnt));     // 重置节点访问次数
        dfs(s, 0);
        if(found) return true;            // 找到了第 k 条路径
        if(next_bound == inf) return false; // 没有更多路径
        bound = next_bound;                // 提高上界，继续迭代
    }
}

signed main() {
    cin.tie(nullptr)->sync_with_stdio(false);
    cin >> n >> m;
    for(int i = 1; i <= m; i++) {
        int u, v, w; cin >> u >> v >> w;
        e[u].push_back({v, w});
        e[v].push_back({u, w});
    }
    cin >> s >> t >> k;
    if(ida_star()) cout << ans << '\n';
    else cout << "-1\n";
    return 0;
}
```
