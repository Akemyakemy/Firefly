## 矩阵相乘

### 缺省源

```c++
const int N=100;//假设矩阵大小为100x100
const int mod=998244353;
typedef int matrix[N][N];
```

### 矩阵乘法

```c++
void matrixMul(matrix& a,matrix& b,int n,int m,int k){//a是n行m列，b是m行k列，c是n行k列
    matrix c;
    memset(c,0,sizeof(matrix));
    for(int i=0;i<n;i++){
        for(int j=0;j<m;j++){
            if(!a[i][j]) continue;
            for(int t=0;t<k;t++){
                if(!b[j][t]) continue;
                c[i][t]=(c[i][t]+1ll*a[i][j]*b[j][t]%mod)%mod;
            }
        }
    }
    memcpy(a,c,sizeof(c));
}
```

### 矩阵快速幂

```c++
void initMatrix(matrix& res){//生成单位矩阵
    memset(res,0,sizeof(matrix));
    for(int i=0;i<N;i++) res[i][i]=1;
}

void qpMatrix(matrix& a,int b){//矩阵快速幂
    matrix res;
    initMatrix(res);
    while(b){
        if(b&1) matrixMul(res,a,N,N,N);
        matrixMul(a,a,N,N,N);
        b>>=1;
    }
    memcpy(a,res,sizeof(matrix));
}
```
