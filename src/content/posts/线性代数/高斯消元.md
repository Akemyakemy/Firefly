## 高斯消元

![](https://img.542000.xyz/file/算法/1781523570932_屏幕截图_2026-06-15_193853.png)

$O(n^3) $

```c++
bool gauss(double a[N][N]){
    for(int i=1;i<=n;i++){//枚举行列
        int r=i;
        for(int k=i;k<=n;k++)
            if(fabs(a[k][i])>eps){r=k;break;}
        if(r^i) swap(a[r],a[i]);//换行
        if(fabs(a[i][i])<eps) return 0;
        
        for(int j=n+1;j>=i;j--)
            a[i][j]/=a[i][i];
        for(int k=i+1;k<=n;k++)
            for(int j=n+1;j>=i;j--)
                a[k][j]-=a[k][i]*a[i][j];
    }
    for(int i=n-1;i>=1;i--)//回代
        for(int j=i+1;j<=n;j++)
            a[i][n+1]-=a[i][j]*a[j][n+1];
    return 1;
}
```

### 高斯-约旦消元法

![](https://img.542000.xyz/file/算法/1781523783281_屏幕截图_2026-06-15_194241.png)

```c++
bool gauss_jordan(double a[N][N]){
    for(int i=1;i<=n;i++){
        int r=i;
        for(int k=i;k<=n;k++)
            if(fabs(a[k][i])>eps){r=k;break;}
        if(r^i) swap(a[r],a[i]);
        if(fabs(a[i][i])<eps) return 0;
        
        for(int k=1;k<=n;k++){//对角化
            if(k==i) continue;
            double t=a[k][i]/a[i][i];
            for(int j=i;j<=n+1;j++)
                a[k][j]-=t*a[i][j];
        }
    }
    for(int i=1;i<=n;i++) a[i][n+1]/=a[i][i];
    return 1;
}
```
