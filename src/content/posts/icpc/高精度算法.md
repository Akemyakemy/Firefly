## 高精度算法

超过 $2^{64}$ 大小的数需要使用高精度算法实现。

可以通过数组模拟实现。

这里收集了常见的大数算法，如果你觉得 $python$ 和 $java$ 够用可以 [$Skip$]([归档 - 我不是0](https://blog.542000.xyz/archive/)) 。

### 加

```c++
void add(int a[],int b[],int c[]){
    memset(c,0,sizeof(c));
    for(int i=0;i<LEN-1;i++){
        c[i]+=a[i]+b[i];
        if(c[i]>=10){//加法最多进一
            c[i+1]+=1;
            c[i]-=10;
        }
    }
}
```

### 乘

#### 单精度

```c++
void mul(int a[],int b,int c[]){
    memset(c,0,sizeof(c));
    for(int i=0;i<LEN-1;i++){
        c[i]+=a[i]*b[i];
        if(c[i]>=10){
            c[i+1]+=c[i]/10;
            c[i]%=10;
        }
    }
}
```

#### 双精度

```c++
void mul(int a[],int b[],int c[]){
    clear(c);
    for(int i=0;i<LEN-1;i++){
        // 位数为i可以拆解为j+(i-j)的位数相乘
        for(int j=0;j<=i;j++) c[i]+=a[j]*b[i-j];
        if(c[i]>=10){
            c[i+1]+=c[i]/10;
            c[i]%=10;
        }
    }
}
```

### 除

竖式长除法

```c++
// 被除数a以下标idx为最低位，是否能减b
// len是b的长度，避免反复计算
bool cmp(int a[],int b[],int idx,int len){
    // 由于竖式长除法从大到小递推计算，此处a的位数比b多的情况只可能为多1位，若该位不为0，则一定可减
    if(!a[idx+len]) return 1;
    
    for(int i=len-1;~i;i--){
        if(a[idx+i]>b[i]) return 1;
        if(a[idx+i]<b[i]) return 0;
    }
    return 1;//两数相等也可减
}

void div(int a[],int b[],int c[],int d[]){
    memset(c,0,sizeof(c));memset(d,0,sizeof(d));//c存商，d存余数
    
    int la=LEN-1,lb=LEN-1;//找a和b的最高位
    while(!a[la-1] && la>0) la--;
    while(!b[lb-1] && lb>0) lb--;
    if(!lb){
        puts("除数不能为0 !!!");
        return;
    }
    
    memcpy(d,a,la*sizeof(int));//除完后留下的部分自然为余数
    for(int i=la-lb;~i;i--){//~i等价于i>=0
        while(cmp(d,b,i,lb)){
            c[i]++;//商的该位加1
            //模拟高精度减法
            for(int j=0;j<lb;j++){
                d[i+j]-=b[j];
                if(d[i+j]<0){
                    d[i+j+1]-=1;
                    d[i+j]+=10;
                }
            }
        }
    }
}
```

### 高精度快速幂

```c++
void qpow(int a[],int b,int c[]){
    memset(c,0,sizeof(c)),c[0]=1;
    while(b){
        if(b&1) c=mul(c,a);//高精度乘法
        a=mul(a,a);
        b>>=1;
    }
}
```

### 二进制转十进制

```c++
int trans(int a[],int b[],int LEN){//LEN为2进制数a的长度
    memset(b,0,sizeof(b));
    int idx=0;//b目前最高位，初始为个位
    
    //reverse(a,a+LEN);//a最高位在后时使用
    for(int i=0;i<LEN;i++){//a的最高位在前，b的最高位在后
        int c=0;//进位标志，初始无进位
        for(int j=0;j<=id;j++){//因为a从高位开始转，所以可以每一次都乘以2来动态转换
            int t=b[j]*2+c;
            b[j]=t%10;
            c=t/10;
        }
        if(c) b[++idx]=c;
        
        if(!a[i]) continue;//当前位为0则不加
        c=1;
        for(int j=0;j<=idx && c;j++){//模拟当前位+1
            int t=b[j]+c;
            b[j]=t%10;
            c=t/10;
        }
        if(c) b[++idx]=c;
    }
    return idx;//返回b的位数
}
```
