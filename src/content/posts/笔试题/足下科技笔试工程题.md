1. 题目

实现一个多线程安全的同步队列, C++

```cpp
template <typename T>
struct SyncQueue {
 public:
  SyncQueue(size_t capacity) : capacity(capacity), interrupt_flag(false) {}
  SyncQueue() = delete;
  SyncQueue(const SyncQueue<T>&) = delete;
  SyncQueue& operator=(const SyncQueue<T>&) = delete;
  ~SyncQueue() {
    interrupt();
  }

  // 唤醒被阻塞的线程并退出
  void interrupt() {
    std::lock_guard<std::mutex> lock(queue_mutex);
    interrupt_flg=true;
    queue_not_full.notify_all();
    queue_not_empty.notify_all();
  }

  // 尝试向队列添加元素，如果失败立即返回
  bool try_push(const T& event) {
    std::lock_guard<std::mutex> lock(queue_mutex);
    if(full() || interrupt_flg) return false;
    queue.push(event);
    queue_not_empty.notify_one();
    return true;
  }

 // 尝试从队列取出一个元素，如果失败立即返回
  std::optional<T> try_pop() {
    std::lock_guard<std::mutex> lock(queue_mutex);
    if(empty() || interrupt_flg) return std::nullopt;
    T val=queue.front();
    queue.pop();
    queue_not_full.notify_one();
    return val;
  }

  // 向队列尾部添加元素，如果队列满则等待
  bool push(const T& element) {
    std::unique_lock<std:mutex> lock(queue_mutex);
    queue_not_full.wait(lock,[this](){
        return interrupt_flg || !full();
    });
    if(interrupt_flg) return false;
    queue.push(element);
    queue_not_empty.notify_one();
    return true;
  }

  // 从队列取出并删除头部元素，如果队列空则等待
  std::optional<T> pop() {
    std::unique_lock<std:mutex> lock(queue_mutex);
    queue_not_empty.wait(lock,[this](){
        return interrupt_flg || !empty();
    });
    if(interrupt_flg) return std::nullopt;
    T val=queue.front();
    queue.pop();
    queue_not_full.notify_one();
    return val;
  }

  size_t length() {
    std::lock_guard<std:mutex> lock(queue_mutex);
    return queue.size();
  }

  bool full() {
    return queue.size()>=capacity;
  }

  bool empty() {
    return queue.empty();
  }

  void clear() {
    std::lock_guard<std::mutex> lock(queue_mutex);
    while(queue.size()) queue.pop();
  }

 private:
  size_t capacity;
  std::atomic<bool> interrupt_flag;
  std::queue<T> queue;
  std::mutex queue_mutex;
  std::condition_variable queue_not_full;
  std::condition_variable queue_not_empty;
};
```

2. 题目
实现一个 Watchdog，要求：
可注册多个模块
每个模块定期调用 feed()
若超过超时时间未 feed，则触发报警回调
内部使用一个监控线程
```cpp
class Watchdog {
public:
    using Callback = std::function<void(const std::string& module)>;

    Watchdog(size_t timeoutMs, Callback cb);
    void feed(const std::string& module);

private:
    // TODO
    size_t timeout_ms;
    Callback alarm_cb;
    std::atomic<bool> running;
    std::thread monitor_thread;
    std::mutex mtx;
    std::unordered_map<std::string,std::chrono::steady_clock::time_point> last_feed_time;
    void monitor_loop();
    ~Watchdog();
};

Watchdog::Watchdog(size_t timeoutMs,Callback cb):timeout_ms_(timeoutMs),alarm_cb_(std::move(cb)),running(true){
    monitor_thread=std::thread(&Watchdog::monitor_loop,this);
}

Watchdog::~Watchdog(){
    running=false;
    if(monitor_thread.joinable()) monitor_thread.join();
}

void Watchdog::feed(const std::string& module){
    std::lock_guard<std::mutex> lock(mxt);
    last_feed_time[module]=std::chrono::steady_clock::now();
}

void Watchdog::monitor_loop(){
    while(running_){
        std::this_thread::sleep_for(std::chrono::milliseconds(100));
        std::lock_guard<std::mutex> lock(mxt);
        auto now=std::chrono::steady_clock::now();
        for(auto& item:last_feed_time){
            const std::string& module_name=item.first;
            auto& last_ts=item.second;
            auto delta=std::chrono::duration_cast<std::chrono::milliseconds>(now-last_ts).count();
            if(delta>static_cast<long long>(timeout_ms)){
                alarm_cb(module_name);
                last_ts=now;
            }
        }
    }
}
```

3. 题目
实现一个 TimerManager，要求：

可注册多个定时器，每个定时器包含：
超时时间（毫秒）
回调函数（可捕获 lambda）
内部采用一个线程负责管理与触发定时器
定时器触发后执行其回调

```cpp
#include <functional>
#include <chrono>

class TimerManager {
public:
    using Callback = std::function<void()>;

    TimerManager();
    ~TimerManager();

    void addTimer(size_t timeoutMs, Callback cb);

private:
    // 你需要哪些成员？
    struct TimerTask{
        std::chrono::steady_clock::time_point expire_time;
        Callback callback;
        bool operator > (const TimerTask& other) const {
            return expire_time>other.expire_time;
        }
    };

    void run_loop();
    std::priority_queue<TimerTask,std::vector<TimerTask>,std::greater<TimerTask>> min_heap;
    std::mutex mtx;
    std::condition_variable cv;
    std::atomic<bool> running;
};

TimerManager::TimerManager():running(true){
    std::thread(&TimerManager::run_loop,this).datch();
}

TimerManager::~TimerManager(){
    running=false; 
    cv.notify_one();
}

void TimerManager::addTimer(size_t timeoutMs,Callback cb){
    std::lock_guard<std::mutex> lock(mxt);
    auto expire=std::chrono::steady_clock::now()+std::chrono::milliseconds(timeoutMs);
    min_heap.push({expire,std::move(cb)});
    cv.notify_one();
}

void TimerManager::run_loop(){
    while(running){
        std::unique_lock<std::mutex> lock(mxt);
        if(min_heap.empty()){
            cv.wait(lock);
            continue;
        }
        auto& top_task=min_heap.top();
        auto now=std::chrono::steady_clock::now();
        auto wait_time=top_task.expire_time-now;
        bool timeout=cv.wait_for(lock,wait_time)==std::cv_status::timeout;
        if(!timeout) continue;
        auto task=top_task;
        min_heap.pop();
        lock.unlock();
        task.callback();
    }
}

// 示例：
int main() {
    TimerManager tm;
    tm.addTimer(1000, [] { printf("1s timer!\n"); });
    tm.addTimer(500, [] { printf("0.5s timer!\n"); });
    std::this_thread::sleep_for(std::chrono::seconds(2));
}
```
