### 链式前向星

本质上是使用联表实现的。

```c++
int head[N],ne[N<<1],to[N<<1],we[N<<1],cnt;

void init(){
    memset(head,-1,sizeof(head));
    memset(ne,-1,sizeof(ne));
    memset(to,-1,sizeof(to));
    memset(we,-1,sizeof(we));
    cnt=0;//cnt存边的编号，从0开始
}

void add(int u,int v,int w){
    ne[cnt]=head[u];
    to[cnt]=v;
    we[cnt]=w;
    head[u]=cnt++;
}

//遍历u的出边
for(int i=head[u];~i;i=ne[i]){
    int v=to[i];
}
```
