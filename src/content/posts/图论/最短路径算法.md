## 最短路径算法

这里只介绍最常用的算法。

### Floyd算法

Floyd算法是一种动态规划算法，用于求解有向图中任意两点之间的最短路径。

$O(n^3) $

```c++
//d[i][j]表示从i到j的最短距离
void floyd(){
    for(int k=1;k<=n;k++){
        for(int i=1;i<=n;i++){
            for(int j=1;j<=n;j++) d[i][j]=min(d[i][j],d[i][k]+d[k][j]);
        }
    }
}
```

#### 传递闭包

Floyd算法同时也是传递 闭包的算法模板。

```c++
//d[i][j]表示i与j是否连通
void floyd(){
    for(int k=1;k<=n;k++){
        for(int i=1;i<=n;i++){
            for(int j=1;j<=n;j++) a[i][j]|=a[i][k]&a[k][j];
        }
    }
}
```

### SPFA算法

是Bellman-Ford算法的改进。基于松弛操作的最短路算法，可以求出**有负权**的图的最短路径，并可以对**最短路径不存在的情况**（判负环）进行判断。

$O(nm)$

#### 队列优化

```c++
const int inf=LLONG_MAX;

struct edge{int v,w;};//终点和边权
vector<edge> e[N];
int dis[N],cnt[N],vis[N];
int pre[N];
queue<int> q;

bool spfa(int s){
    for(int i=1;i<=n;i++) dis[i]=inf;
    dis[s]=0;vis[s]=1;q.push(s);
    while(q.size()){
        int u=q.front();q.pop();vis[u]=0;
        for(auto ed:e[u]){
            int v=ed.v,w=ed.w;
            if(dis[v]>dis[u]+w){
                dis[v]=dis[u]+w;
                cnt[v]=cnt[u]+1;
                pre[v]=u;
                if(cnt[v]>=n) return true;//负负相加，存在负环
                if(!vis[v]) vis[v]=1,q.push(v);//v点被更新且不在队内，入队
            }
        }
    }
    return false;
}

void dfs_path(int u){//递归输出最短路径
    if(u==s){
        printf("%d ",u);
        return;
    }
    dfs_path(pre[u]);
    printf("%d ",u);
}
```

#### 递归SPFA

这个写起来十分简单，但是只有判断负环是否存在的功能。

```c++
bool spfa(int u){
    vis[u]=1;
    for(int i=h[u];~i;i=ne[i]){//链式前向星
        int v=to[i];
        if(d[v]>d[u]+w[i]){
            d[v]=d[u]+w[i];
            pre[v]=u;
            if(vis[v]||spfa(v)) return true;//如果v已经过（找到环）或递归发现负环
        }
    }
    vis[u]=0;
    return false;
}
```

### Dijkstra算法

无环图单源最短路径。

$O(mlogn)$

```c++
#define pii pair<int,int>
#define x first   //dis 距离
#define y second  //u   节点编号
priority_queue<pii,vector<pii>,greater<pii>> q;//小根堆，按dis排序

struct edge{
    int v,w;
}；
vector<edge> e[N];
int dis[N],vis[N];

void dijkstra(int s){
    for(int i=1;i<=n;i++) dis[i]=inf;
    memset(vis,0,(n+1)*sizeof(int));
    dis[s]=0;q.push({0,s});

    while(q.size()){
        int u=q.top().y;q.pop();
        if(vis[u]) continue;
        vis[u]=1;
        for(auto ed:e[u]){
            int v=ed.v,w=ed.w;
            if(dis[v]>dis[u]+w){
                dis[v]=dis[u]+w;
                q.push({dis[v],v});
            }
        }
    }
}
```

### Johnson全源最短路径算法

$\text{1. 新建一个虚拟源点 } 0\text{，从该点向其他所有点连一条边权为 } 0 \text{ 的边，再用 } \text{SPFA} \text{ 算法求出从 } 0 \text{ 号点到其他所有点的最短路 } h(i)\text{。}$

$\text{2. 将新图的边权改造为 } w(u,v) + h(u) - h(v)\text{，这样能确保边权非负。}$

$\text{3. 以每个点为起点，跑 } n \text{ 轮 } \text{Heap-Dijkstra} \text{ 算法，求出任意两点间最短路。}$

$O(nmlogm)$

```c++
#define pii pair<int,int>
#define x first
#define y second
struct edge{int v,w;};
vector<edge> e[N];
int vis[N],cnt[N];
int h[N],d[N];

bool spfa();
void dijkstra();

int main(){
    cin>>n>>m;
    for(int i=0;i<m;i++) cin>>a>>b>>c,e[a].push_back({b,c});
    for(int i=1;i<=n;i++) e[0].push_back({i,0});
    
    if(!spfa()){
        for(int u=1;u<=n;u++)
            for(auto &ed:e[u])
                ed.w+=h[u]-h[ed.v];//构造新边
        for(int i=1;i<=n;i++) dijkstra(i);//每次的结果都会被刷新，没有做存储
    }
    return 0;
}

bool spfa(){
    queue<int> q;
    for(int i=1;i<=n;i++) h[i]=inf;
    memset(vis,0,(n+1)*sizeof(vis));
    h[0]=0,vis[0]=1;q.push(0);
    while(q.size()){
        int u=q.front();q.pop();vis[u]=0;
        for(auto ed:e[u]){
            int v=ed.v,w=ed.w;
            if(h[v]>h[u]+w){
                h[v]=h[u]+w;
                cnt[v]=cnt[u]+1;
                if(cnt[v]>=n) return true;
                if(!vis[v]) q.push(v),vis[v]=1;
            }
        }
    }
    return false;
}

void dijkstra(int s){
    priority_queue<pii,vector<pii>,greater<pii>> q;
    for(int i=1;i<=n;i++) d[i]=inf;
    memset(vis,0,(n+1)*sizeof(int));
    d[s]=0;q.push({0,s});
    while(q.size()){
        int u=q.top().y;q.pop();
        if(vis[u]) continue;
        vis[u]=1;
        for(auto ed:e[u]){
            int v=ed.v,w=ed.w;
            if(d[v]>d[u]+w){
                d[v]=d[u]+w;
                if(!vis[v]) q.push({d[v],,v});
            }
        }
    }
}
```
