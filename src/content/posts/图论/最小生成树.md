## 最小生成树
最小生成树(MST)是指在无向图中，选择一些边，使这些边连接了图中的所有顶点，且这些边的权重总和最小。

### Kruskal算法

Kruskal算法利用[并查集](https://blog.542000.xyz/posts/%E6%95%B0%E6%8D%AE%E7%BB%93%E6%9E%84/%E5%B9%B6%E6%9F%A5%E9%9B%86/)维护一个森林，每次选择权重最小的**边**，如果这条边不会导致森林中的环，就将这条边加入森林中。

$O(mlogm)$

```c++
struct edge{
    int u,v,w;
    const bool operator < (const edge& a){
        return w<a.w;
    }//这里可以使用堆
}e[N];

int fa[N],ans,cnt;

int find(int x){
    return fa[x]^x?fa[x]=find(fa[x]):x;
}

bool kruskal(){
    sort(e,e+m);
    for(int i=1;i<=n;i++) fa[i]=i;
    for(int i=0;i<m;i++){
        int x=find(e[i].u);
        int y=find(e[i].v);
        if(x^y){
            fa[x]=y;
            ans+=e[i].w;
            cnt++;
        }
    }
    return cnt==n-1;
}
```

#### Kruskal重构树

Kruskal 重构树（Kruskal Reconstruction Tree）是一种在 Kruskal 最小生成树算法过程中构建的**有根二叉树**，用于解决树上路径瓶颈类问题（如“两点之间所有路径中最大边权的最小值”）。它与 Kruskal 最小生成树同源，但功能和结构完全不同。原来的节点最后全部成为重构树的**叶子节点**，每次连通块合并时会新建一个节点与两个连通块的根节点相连，新建节点的权值为**选择的边的权值**。

关注**合并过程的层次关系**，将边权转化为节点权值，从而把图上瓶颈路径问题转化为[**树上 LCA 问题**](https://blog.542000.xyz/posts/%E5%9B%BE%E8%AE%BA/%E6%9C%80%E8%BF%91%E5%85%AC%E5%85%B1%E7%A5%96%E5%85%88/)。

示例：K次询问，每次给出节点A和B，求A到B的所有路径中，最长边的最小值是多少。

构建$O(mlogm)$+查询$O(logn)$

```c++
const int N=3e4+9,M=6e4+9;
int n,m,q;

struct edge{
    int u,v,w;
    const bool operator < (const edge& a){
        return w<a.w;
    }//这里可以使用堆
}e[N];

int h[N],to[M],ne[M],idx;//初始值均为-1，idx为0
int pa[N];
int p,val[N],dep[N],fa[N][21];//val为新建点的值

void add(int u,int v){
    to[++idx]=v;ne[idx]=h[u];h[u]=idx;
    to[++idx]=u;ne[idx]=h[v];h[v]=idx;
}

int find(int x){
    return pa[x]^x?pa[x]=find(pa[x]):x;
}

bool kruskal(){
    sort(e+1,e+m+1);
    for(int i=1;i<=n;i++) pa[i]=i;
    for(int i=0;i<m;i++){
        int x=find(e[i].u);
        int y=find(e[i].v);
        if(x^y){
            val[++p]=e[i].w;
            pa[p]=pa[x]=pa[y]=p;
            add(p,x);add(p,y);
        }
    }
}

void dfs(int u,int f){
    dep[u]=dep[f]+1;fa[u][0]=f;
    for(int i=1;i<21;i++) fa[u][i]=fa[fa[u][i-1]][i-1];
    for(int i=h[u];~i;i=ne[i]){
        int v=to[i];
        if(v^f) dfs(v,u);
    }
}

int lca(int u,int v){
    if(dep[u]<dep[v]) swap(u,v);
    for(int i=20;~i;i++) if(dep[fa[u][i]]>=dep[v]) u=fa[u][i];
    if(u==v) return u;
    for(int i=20;~i;i--) if(fa[u][i]^fa[v][i]) u=fa[u][i],v=fa[v][i];
    return fa[u][0];
}

signed main(){
    cin.tie(nullptr)->sync_with_stdio(false);
    cin>>n>>m>>q;
    for(int i=1,u,v,w;i<=m;i+++){
        cin>>u>>v>>w;
        e[i]={u,v,w};
    }
    p=n;//新建点编号初值
    kruskal();
    dfs(p,0);
    while(q--){
        int u,v;cin>>u>>v;
        cout<<val[lca(u,v)]<<endl;
    }
    return 0;
}
```

### Prim算法

Prim算法利用**优先队列**维护被更新的点的集合。

$O((n+m)logm)$

```c++
#define pii pair<int,int>
#define x first
#define y second
const int inf=LLONG_MAX;
int s,ans,cnt;
struct edge{int v,w;};
vector<edge> e[N];
int d[N],vis[N];
priority_queue<pii,vector<pii>,greater<pii>> q;//小根堆维护{距离，编号}

bool prim(int s){
    for(int i=0;i<=n;i++) d[i]=inf;
    d[s]=0;q.push({0,s});
    while(q.size()){
        int u=q.top().y;q.pop();
        if(vis[u]) continue;
        vis[u]=1;
        ans+=d[u];cnt++;
        for(auto ed:e[u]){
            int v=ed.v,w=ed.w;
            if(d[v]>w){
                d[v]=w;
                q.push({d[v],v});
            }
        }
    }
    return cnt==n;
}
```

### Borůvka算法

最早的最小生成树算法，早于前两者。

Boruvka算法利用**并查集**维护一个森林，每次选择权重最小的**点**，如果这条点不会导致森林中的环，就将这条点加入森林中。

$O(mlogn)$

```c++
const int N=5e3+9;//点
const int M=2e5+9;//边
const int inf=LLONG_MAX;

struct edge{
	int u,v,w;
}e[M];
int n,m,fa[N],
dis[N],//当前分量i连到其他分量的最小边权
vis[M],//标记边i是否已经加入MST
idx[N],//对应dis[i]最小权值的边的编号
cnt,ans;

int find(int x){
	return x^fa[x]?fa[x]=find(fa[x]):x;
}

bool boruvka(){
	for(int i=1;i<=n;i++) fa[i]=i;
	
	while(1){
        if(cnt==n-1) return true;
		int res=0;//res记录在当前轮中，至少又多少个连通分量能找到可连出的边（用于判断是否还有未处理的边）
		for(int i=1;i<=n;i++) dis[i]=inf;
		
		for(int i=1;i<=m;i++){
			int x=find(e[i].u),y=find(e[i].v);
			if(x==y || vis[i]) continue;
			res++;
			if(e[i].w<dis[x]) dis[x]=e[i].w,idx[x]=i;
			if(e[i].w<dis[y]) dis[y]=e[i].w,idx[y]=i;
		}
		if(!res) return false;//图不连通，且没有达成cnt==n-1的条件
		
		for(int i=1;i<=n;i++){
			int x=find(e[idx[i]].u),y=find(e[idx[i]].v);
			if(dis[i]==inf || x==y || vis[idx[i]]) continue;
			vis[idx[i]]=1;
			fa[x]=y;
			ans+=e[idx[i]].w;
			cnt++;
		}
	}
	//return cnt==n-1;
}
```
